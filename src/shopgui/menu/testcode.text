                                leftover_item_meta = leftover_item.item_meta

                                leftover_item.set_item_meta(leftover_item_meta)
                                player.world.drop_item(player.location, leftover_item)
                                
                            menu.close(player)
                    except Exception:
                        # Phương án dự phòng nếu hàm add_item của bro không trả về leftovers (Cơ chế add cổ điển)
                        # Bro có thể check nhanh bằng số ô trống trước khi add

                        else:
                            player.send_message("§c Your inventory is so full, items have been dropped!")
                            player.world.drop_item(player.location, give_item)
                        menu.close(player)
                
            return
